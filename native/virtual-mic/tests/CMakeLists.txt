cmake_minimum_required(VERSION 3.22)
project(JinkraVadPortableTests LANGUAGES CXX)
enable_testing()

function(jinkra_enable_runtime_assertions target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
endfunction()

add_library(jinkra_audio_core STATIC ../shared/jinkra_audio_core.cpp ../shared/jinkra_cable_pipeline.cpp)
target_include_directories(jinkra_audio_core PUBLIC ../shared)
target_compile_features(jinkra_audio_core PUBLIC cxx_std_20)
add_executable(core_tests core_tests.cpp)
target_link_libraries(core_tests PRIVATE jinkra_audio_core)
jinkra_enable_runtime_assertions(core_tests)
add_test(NAME core_tests COMMAND core_tests)
add_executable(pipeline_tests pipeline_tests.cpp)
target_link_libraries(pipeline_tests PRIVATE jinkra_audio_core)
jinkra_enable_runtime_assertions(pipeline_tests)
add_test(NAME pipeline_tests COMMAND pipeline_tests)
